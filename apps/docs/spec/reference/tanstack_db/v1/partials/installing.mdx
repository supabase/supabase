### Install as a package

<RefSubLayout.EducationRow>
  <RefSubLayout.Details>

    Install `@supabase-labs/tanstack-db` alongside `@tanstack/react-db` and your existing `@supabase/supabase-js` client.

  </RefSubLayout.Details>

  <RefSubLayout.Examples>

    <Tabs
      size="small"
      type="underlined"
      defaultActiveId="npm"
      queryGroup="platform"
    >
      <TabPanel id="npm" label="npm">

        ```sh Terminal
        npm install @supabase-labs/tanstack-db @tanstack/react-db
        ```

      </TabPanel>
      <TabPanel id="yarn" label="Yarn">

        ```sh Terminal
        yarn add @supabase-labs/tanstack-db @tanstack/react-db
        ```

      </TabPanel>
      <TabPanel id="pnpm" label="pnpm">

        ```sh Terminal
        pnpm add @supabase-labs/tanstack-db @tanstack/react-db
        ```

      </TabPanel>
    </Tabs>

  </RefSubLayout.Examples>
</RefSubLayout.EducationRow>
