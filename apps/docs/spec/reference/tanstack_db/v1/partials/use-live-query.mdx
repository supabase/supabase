title: Live queries and mutations
---

Read a collection reactively with `useLiveQuery()`, and mutate it optimistically. The UI updates immediately, and changes roll back if the server rejects them.

<RefSubLayout.EducationRow>
  <RefSubLayout.Details>

    `useLiveQuery` re-runs whenever the underlying collection changes, so components stay in sync without manual cache invalidation. `.insert()`, `.update()`, and `.delete()` apply to the UI immediately and roll back if the server rejects the change.

  </RefSubLayout.Details>

  <RefSubLayout.Examples>

    ```tsx
    import { useLiveQuery, eq } from '@tanstack/react-db'
    // import the todos collection you defined
    import { todos } from './collections'

    const { data, isLoading } = useLiveQuery(
      (q) => q.from({ todo: todos }).where(({ todo }) => eq(todo.completed, false)),
      []
    )

    const insertTodo = (title: string = 'Buy milk') => {
      todos.insert({ id, title: title, completed: false })
    }

    const updateTodo = (id: string) => {
      todos.update(id, (draft) => {
        draft.completed = true
      })
    }

    const deleteTodo = (id: string) => {
      todos.delete(id)
    }
    ...
    ```

  </RefSubLayout.Examples>
</RefSubLayout.EducationRow>
